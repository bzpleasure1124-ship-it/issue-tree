        function generateId() { return 'node_' + Math.random().toString(36).substr(2, 9); }

        function findNodeInfo(node, id, parent = null, index = 0, depth = 0) {
            if (node.id === id) return { node, parent, index, depth };
            for (let i = 0; i < node.children.length; i++) {
                const result = findNodeInfo(node.children[i], id, node, i, depth + 1);
                if (result) return result;
            }
            return null;

        // --- ロジック関数群 ---
        function actionAddBelow(nodeId, currentText) {
            const info = findNodeInfo(treeData, nodeId);
            if (!info) return;
            info.node.text = currentText;
            const newNode = { id: generateId(), text: '', isRed: false, children: [] };
            if (info.parent) info.parent.children.splice(info.index + 1, 0, newNode);
            else info.node.children.push(newNode);
            selectedNodeId = newNode.id; saveData(); renderTree();
        }

        function actionAddRight(nodeId, currentText) {
            const info = findNodeInfo(treeData, nodeId);
            if (!info) return;
            info.node.text = currentText;
            const newNode = { id: generateId(), text: '', isRed: false, children: [] };
            info.node.children.push(newNode);
            selectedNodeId = newNode.id; saveData(); renderTree();
        }

        function actionMoveUp(nodeId) {
            const info = findNodeInfo(treeData, nodeId);
            if (!info || !info.parent) return;
            const siblings = info.parent.children;
            if (info.index > 0) {
                const temp = siblings[info.index];
                siblings[info.index] = siblings[info.index - 1];
                siblings[info.index - 1] = temp;
                saveData(); renderTree();
            }
        }

        function actionMoveDown(nodeId) {
            const info = findNodeInfo(treeData, nodeId);
            if (!info || !info.parent) return;
            const siblings = info.parent.children;
            if (info.index < siblings.length - 1) {
                const temp = siblings[info.index];
                siblings[info.index] = siblings[info.index + 1];
                siblings[info.index + 1] = temp;
                saveData(); renderTree();
            }
        }

        function actionToggleRed(nodeId) {
            const info = findNodeInfo(treeData, nodeId);
            if (!info) return;
            info.node.isRed = !info.node.isRed;
            saveData(); renderTree();
        }

        function actionDelete(nodeId) {
            const info = findNodeInfo(treeData, nodeId);
            if (!info) return;
            if (!info.parent) {
                showAlert("トップイシュー自体は削除できません。\n文字を書き換えてください。");
                return;
            }

            const executeDelete = () => {
                info.parent.children.splice(info.index, 1);
                selectedNodeId = info.index > 0 ? info.parent.children[info.index - 1].id : info.parent.id;
                saveData(); renderTree();
            };

            if (info.node.children.length > 0) {
                showConfirm("この箱には子の枝が付いています。\n枝もすべて消去しますがよろしいですか？", executeDelete);
            } else {
                executeDelete();
            }
        }

        function handleKeyDown(e, node, textDiv) {
            if (e.key === 'Enter') { e.preventDefault(); actionAddBelow(node.id, textDiv.innerText); }
            else if (e.key === 'Tab') { e.preventDefault(); actionAddRight(node.id, textDiv.innerText); }
            else if (e.altKey && e.key === 'ArrowUp') { e.preventDefault(); actionMoveUp(node.id); }
            else if (e.altKey && e.key === 'ArrowDown') { e.preventDefault(); actionMoveDown(node.id); }
            else if ((e.ctrlKey || e.metaKey) && (e.key === 'r' || e.key === 'R')) { e.preventDefault(); actionToggleRed(node.id); }
            else if ((e.key === 'Delete' || e.key === 'Backspace')) {
                const selection = window.getSelection();
                const isAllSelected = selection.toString() === textDiv.innerText;
                const isEmpty = textDiv.innerText.trim() === '';
                if (isEmpty || isAllSelected) { 
                    e.preventDefault(); 
                    actionDelete(node.id); 
                }
            }
        }

        const RED_MARKER = "★ ";

        function exportExcel() {
            const excelData = [];
            const merges = []; 

            function traverse(node, depth, rowIndex) {
                if (!excelData[rowIndex]) excelData[rowIndex] = [];
                
                let cellText = node.text;
                if (node.isRed) cellText = RED_MARKER + cellText;

                excelData[rowIndex][depth] = {
                    v: cellText,
                    t: 's',
                    s: {
                        alignment: { vertical: "center", horizontal: "left", wrapText: true },
                        font: { color: { rgb: node.isRed ? "E53E3E" : "333333" }, bold: node.isRed },
                        border: {
                            top: { style: "thin", color: { rgb: "E2E8F0" } },
                            bottom: { style: "thin", color: { rgb: "E2E8F0" } },
                            left: { style: "thin", color: { rgb: "E2E8F0" } },
                            right: { style: "thin", color: { rgb: "E2E8F0" } }
                        }
                    }
                };

                if (!node.children || node.children.length === 0) return rowIndex + 1;

                let currentRow = rowIndex;
                for (let i = 0; i < node.children.length; i++) {
                    currentRow = traverse(node.children[i], depth + 1, currentRow);
                }

                if (currentRow - 1 > rowIndex) {
                    merges.push({
                        s: { r: rowIndex, c: depth },
                        e: { r: currentRow - 1, c: depth }
                    });
                }
                return currentRow;
            }

            const headerRow = [];
            for (let i = 0; i <= maxDepth; i++) {
                headerRow.push({
                    v: i === 0 ? 'トップイシュー' : `${i}層目`,
                    t: 's',
                    s: {
                        fill: { fgColor: { rgb: "F1F3F5" } },
                        font: { bold: true, color: { rgb: "495057" } },
                        alignment: { horizontal: "center", vertical: "center" },
                        border: {
                            top: { style: "thin", color: { rgb: "CED4DA" } },
                            bottom: { style: "thin", color: { rgb: "CED4DA" } },
                            left: { style: "thin", color: { rgb: "CED4DA" } },
                            right: { style: "thin", color: { rgb: "CED4DA" } }
                        }
                    }
                });
            }
            excelData.push(headerRow);
            traverse(treeData, 0, 1); 

            const ws = XLSX.utils.aoa_to_sheet(excelData);

            if (ws['!ref']) {
                const range = XLSX.utils.decode_range(ws['!ref']);
                for (let R = range.s.r; R <= range.e.r; ++R) {
                    for (let C = range.s.c; C <= range.e.c; ++C) {
                        const cellRef = XLSX.utils.encode_cell({c:C, r:R});
                        if(!ws[cellRef]) {
                            ws[cellRef] = {
                                t: 's', v: '',
                                s: {
                                    border: {
                                        top: { style: "thin", color: { rgb: "E2E8F0" } },
                                        bottom: { style: "thin", color: { rgb: "E2E8F0" } },
                                        left: { style: "thin", color: { rgb: "E2E8F0" } },
                                        right: { style: "thin", color: { rgb: "E2E8F0" } }
                                    }
                                }
                            };
                        }
                    }
                }
            }

            if (merges.length > 0) ws['!merges'] = merges;
            
            const colWidths = [];
            for (let i = 0; i <= maxDepth; i++) colWidths.push({ wch: 35 }); 
            ws['!cols'] = colWidths;

            const wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "イシューツリー");
            XLSX.writeFile(wb, `イシューツリー_${new Date().toISOString().slice(0,10).replace(/-/g, '')}.xlsx`);
        }

        // =========================================================
        // 読込システム

        function importExcel(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                const data = new Uint8Array(e.target.result);
                const workbook = XLSX.read(data, {type: 'array'});
                const jsonData = XLSX.utils.sheet_to_json(workbook.Sheets[workbook.SheetNames[0]], {header: 1});
                
                if(jsonData.length <= 1) return showAlert("データが見つかりません。");

                try {
                    treeData = buildTreeFromArray(jsonData.slice(1));
                    selectedNodeId = treeData.id;
                    saveData(); 
                    renderTree();
                    showAlert("Excelファイルの読込が完了しました。");
                } catch(error) {
                    console.error(error);
                    showAlert("読込に失敗しました。\nこのツールで書き出したファイルを使用してください。");
                }
                event.target.value = "";
            };
            reader.readAsArrayBuffer(file);
        }

        function buildTreeFromArray(rows) {
            let rootNode = null;
            const parentStack = []; 

            for (let r = 0; r < rows.length; r++) {
                const row = rows[r];
                for (let c = 0; c < row.length; c++) {
                    if (row[c] != null && row[c].toString().trim() !== "") {
                        let text = row[c].toString();
                        let isRed = text.startsWith(RED_MARKER);
                        if (isRed) text = text.substring(RED_MARKER.length);

                        const newNode = { id: generateId(), text: text, isRed: isRed, children: [] };
                        parentStack[c] = newNode;

                        if (c === 0) {
                            if (!rootNode) rootNode = newNode; 
                        } else {
                            const parent = parentStack[c - 1];
                            if (parent) {
                                parent.children.push(newNode);
                            }
                        }
                    }
                }
            }
            return rootNode || { id: 'root', text: 'トップイシュー', isRed: false, children: [] };
        }
